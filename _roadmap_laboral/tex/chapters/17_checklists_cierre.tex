\chapter{Checklists de cierre por activo}

\section{Portfolio checklist}


\begin{itemize}
\item Hero comunica Unity Technical Artist / Real-Time 3D.
\item TwinSight aparece primero.
\item Hay demo/video/GIF visible.
\item CTA a GitHub, LinkedIn, CV y contacto.
\item No hay placeholders.
\item Mobile y desktop se leen bien.
\item Limitaciones explicadas.
\end{itemize}


\section{TwinSight checklist}


\begin{itemize}
\item Problem claro.
\item Solution clara.
\item Role propio claro.
\item Features con media.
\item Pipeline CAD-to-realtime.
\item Optimization before/after.
\item Evaluation/metrics verificadas.
\item Limitations sin overclaim.
\item Next steps production-ready.
\end{itemize}


\section{GitHub checklist}


\begin{itemize}
\item TwinSight pinned primero.
\item README con demo, features, pipeline, setup y limitations.
\item Screenshots cargan.
\item No secretos.
\item No repos irrelevantes arriba.
\item ARA solo si funcional.
\end{itemize}


\section{CV checklist}


\begin{itemize}
\item Una pagina.
\item Titulo Unity Technical Artist / Real-Time 3D Developer.
\item Links vivos.
\item TwinSight primer proyecto.
\item Bullets con tecnologia y resultado.
\item No Senior como titulo.
\item No cursos sin certificado como certificaciones.
\end{itemize}


\section{LinkedIn checklist}


\begin{itemize}
\item Headline alineado.
\item About corto y tecnico.
\item Featured con TwinSight/demo/GitHub.
\item Experience/project entry para TwinSight.
\item Skills ordenadas.
\item Availability Colombia remote contractor/B2B.
\end{itemize}


\section{ArtStation checklist}


\begin{itemize}
\item Cover fuerte.
\item Breakdown, no solo renders.
\item Pipeline visual.
\item Human CGCookie como apoyo visual.
\item Links a portfolio/GitHub.
\end{itemize}


\section{Demo reel checklist}


\begin{itemize}
\item 90 segundos main.
\item 15-30 segundos teaser.
\item Hook en primeros 8 segundos.
\item Features visibles.
\item Pipeline/metrics aparecen.
\item CTA final.
\end{itemize}


\section{ARA checklist}


\begin{itemize}
\item README honesto.
\item Demo minima reproducible.
\item No production AI platform.
\item Arquitectura simple.
\item Limitaciones claras.
\end{itemize}
